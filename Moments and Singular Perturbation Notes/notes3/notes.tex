\section{Linear Systems - Thinking about explicit solutions and moments}
Consider the following linear time-invariant, SISO system:
\label{sec:linear-systems}
\begin{equation}
\begin{aligned}
\dot{x} &= A x + B u, \\
y & = Cx,
\end{aligned}
\end{equation}

\begin{equation}
\begin{aligned}
\dot{w} &= Sw, \\
u &= Lw,
\end{aligned}
\end{equation}
Given initial conditions \(x(0) = x_0\) and \(w(0) = w_0\), the solution to the system is given by
\begin{equation}
x(t) = e^{At} x_0 + \int_0^t e^{A(t-\tau)} B L e^{S\tau} w_0 d\tau.
\end{equation}
The output of the system is given by
\begin{equation}
y(t) = C e^{At} x_0 + \int_0^t C e^{A(t-\tau)} B L e^{S\tau} w_0 d\tau.
\end{equation}
If $\sigma(A) \cup \sigma(S) = \emptyset$, then there exists a unique solution $\Pi$ to the Sylvester equation
\begin{equation}
A \Pi + B L = \Pi S.
\end{equation}
Using this, we can substitute an expression for \(B L\) into the integral term for $x(t)$:
\begin{equation}
\begin{aligned}
x(t) &= e^{At} x_0 + \int_0^t e^{A(t-\tau)} (\Pi S - A \Pi) e^{S\tau} w_0 d\tau \\
&= e^{At} x_0 + \int_0^t e^{A(t-\tau)} \Pi S e^{S\tau} w_0 d\tau - \int_0^t e^{A(t-\tau)} A \Pi e^{S\tau} w_0 d\tau \\
&= e^{At} x_0 + \left( \Pi e^{St} - e^{At} \Pi \right) w_0 .
\end{aligned}
\end{equation}
Forward Moments are defined to be 
\begin{equation}
M_g = C \Pi
\end{equation}


\section{Problem Setup}
\label{sec:problem-setup}
Consider the following linear singularly perturbed system:
\begin{equation}
\begin{aligned}
\dot{x}_1(t) &= A_{11} x_1(t) + A_{12} x_2(t) + B_1 u(t), \\
\varepsilon \dot{x}_2(t) &= A_{21} x_1(t) + A_{22} x_2(t) + B_2 u(t),
\end{aligned}
\end{equation}
where \(x_1(t) \in \mathbb{R}^{n_1}\), \(x_2(t) \in \mathbb{R}^{n_2}\), \(u(t) \in \mathbb{R}^m\), and \(0 < \epsilon \ll 1\).

You can write this as a descriptor system: 
\begin{equation}
    E_1 \dot{x} = A x + B u
\end{equation}
where
\begin{equation}
    E_1 = \begin{bmatrix}
    I_{n_1} & 0 \\
    0 & \varepsilon I_{n_2}
    \end{bmatrix}, \quad
    A = \begin{bmatrix}
    A_{11} & A_{12} \\
    A_{21} & A_{22}
    \end{bmatrix}, \quad
    B = \begin{bmatrix}
    B_1 \\
    B_2
    \end{bmatrix}, \quad
    x = \begin{bmatrix}
    x_1 \\
    x_2
    \end{bmatrix}.
\end{equation}
The input signal is also a linear singularly perturbed system: 
\begin{equation}
\begin{aligned}
\dot{w_1} = S_{11} w_1 + S_{12}w_2 \\
\varepsilon \dot{w_2} = S_{21} w_1 + S_{22} w_2
\end{aligned}
\end{equation}
where \(w_1(t) \in \mathbb{R}^{r_1}\), \(w_2(t) \in \mathbb{R}^{r_2}\), and \(0 < \epsilon \ll 1\). 
The input to the system is given by \(u(t) = L_1 w_1(t) + L_2 w_2(t)\).

The input signal can also be written as a descriptor system like so:
\begin{equation}
    E_2 \dot{w} = S w
\end{equation}
where
\begin{equation}
    E_2 = \begin{bmatrix}
    I_{r_1} & 0 \\
    0 & \varepsilon I_{r_2}
    \end{bmatrix}, \quad
    S = \begin{bmatrix}
    S_{11} & S_{12} \\
    S_{21} & S_{22}
    \end{bmatrix}, \quad
    w = \begin{bmatrix}
    w_1 \\
    w_2
    \end{bmatrix}.
\end{equation}
The output of the system is given by
\begin{equation}
    y(t) = C_1 x_1(t) + C_2 x_2(t).
\end{equation}

\section{Wong Sequences}
We define the Wong Sequences such: 
\begin{equation}
    \mathcal{V}_0 = \mathbb{R}^n \qquad \mathcal{V}_{i+1} = A^{-1} (E \mathcal{V}_i) \qquad \mathcal{V}^* = \bigcap_{i=0}^n \mathcal{V}_i
\end{equation}

\begin{equation}
    \mathcal{W}_0 = \{0 \} \qquad \mathcal{W}_{i+1} = E^{-1} (A \mathcal{W}_i) \qquad \mathcal{W}^* = \bigcup_{i=0}^n \mathcal{W}_i
\end{equation}

\section{Relation to Scarciotti Paper}
Suppose the pencil $(S, E_2)$ is regular. 
Then we can write the signal generator as the following: 
\begin{equation}
    \dot{w} = \Theta_2 S w
\end{equation}

Then, we can write the system as:
\begin{equation}
    \begin{aligned}
        E_1 \dot{x} &= A x + B u \\
        \dot{w} &= \Theta_2 S w \\
        y &= Cx \\
        u &= Lw
    \end{aligned}
\end{equation}

So, we can write the corresponding analagous Sylvester Equation: 
\begin{equation}
    A \Pi + B L = E_1 \Pi \Theta_2 S
\end{equation}

and your forward moment becomes: 
\begin{equation}
    M_g = C \Pi
\end{equation}

From here, the results follow from Scarciotti's paper. 

\subsection{When \texorpdfstring{$S_{11} = S_{22} = 0$}{S11 = S22 = 0}}
The pencil $(E_2, S)$ will be regular in this case when: 
\begin{equation}
    \det( \lambda E_2 - S) \neq 0
\end{equation}
for some $\lambda$. Ie:
\begin{equation}
    \det \left(
    \begin{bmatrix}
        \lambda I & -S_{12} \\
        -S_{21} & 0
    \end{bmatrix}
    \right)
    \neq
    0
\end{equation}

Using Schur's Complement formula: 
\begin{equation}
\begin{aligned}
    \det 
    \begin{bmatrix}
        \lambda I & -S_{12} \\
        -S_{21} & 0
    \end{bmatrix}
    &= \det(\lambda I) \det(0 - (-S_{21}) (\lambda I)^{-1} (-S_{12})) \\
    &= \lambda ^ n \det( - \frac{1}{\lambda} S_{21} S_{12}) \\
\end{aligned}
\end{equation}

Thus, regularity requires $S_{21} S_{12}$ to be invertible, and both n x n matrices. 
Which means both $S_{12}, S_{21}$ need to be invertible. 

This means that the system: 
\begin{equation}
    \begin{bmatrix}
        \dot{w}_1 \\
        0
    \end{bmatrix}
    =
    \begin{bmatrix}
        0 & S_{12} \\
        S_{21} & 0
    \end{bmatrix}
    \begin{bmatrix}
        w_1 \\
        w_2
    \end{bmatrix}
\end{equation}

will have no dynamics. 
\begin{equation}
    0 = S_{21} w_1 \implies w_1(t) = 0 \implies w_2 (t) = 0
\end{equation}
This is all when considering the explicit case $\varepsilon = 0$.
